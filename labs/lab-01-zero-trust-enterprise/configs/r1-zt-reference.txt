! LAB-ONLY REFERENCE CONFIGURATION
! Author: mari-2215
! Credentials below must never be reused outside this isolated lab.

enable
configure terminal
hostname R1-ZT
no ip domain-lookup

interface gigabitEthernet0/0
 description TRUNK-TO-SW-CORE
 no ip address
 no shutdown

interface gigabitEthernet0/0.10
 description VLAN10-USERS
 encapsulation dot1Q 10
 ip address 10.10.10.1 255.255.255.0
 ip nat inside
 ip access-group USERS-IN in

interface gigabitEthernet0/0.20
 description VLAN20-FINANCE
 encapsulation dot1Q 20
 ip address 10.10.20.1 255.255.255.0
 ip nat inside

interface gigabitEthernet0/0.30
 description VLAN30-SERVERS
 encapsulation dot1Q 30
 ip address 10.10.30.1 255.255.255.0
 ip nat inside

interface gigabitEthernet0/0.40
 description VLAN40-IT
 encapsulation dot1Q 40
 ip address 10.10.40.1 255.255.255.0
 ip nat inside

interface gigabitEthernet0/0.50
 description VLAN50-GUEST
 encapsulation dot1Q 50
 ip address 10.10.50.1 255.255.255.0
 ip nat inside
 ip access-group GUEST-IN in

interface gigabitEthernet0/0.99
 description VLAN99-MANAGEMENT
 encapsulation dot1Q 99
 ip address 10.10.99.1 255.255.255.0
 ip nat inside

interface gigabitEthernet0/0.999
 description NATIVE-BLACKHOLE
 encapsulation dot1Q 999 native
 no ip address

interface gigabitEthernet0/1
 description EXTERNAL-LAB-SEGMENT
 ip address 198.51.100.1 255.255.255.0
 ip nat outside
 no shutdown

ip dhcp excluded-address 10.10.10.1 10.10.10.20
ip dhcp excluded-address 10.10.20.1 10.10.20.20
ip dhcp excluded-address 10.10.40.1 10.10.40.20
ip dhcp excluded-address 10.10.50.1 10.10.50.20

ip dhcp pool USERS
 network 10.10.10.0 255.255.255.0
 default-router 10.10.10.1
 dns-server 10.10.30.10

ip dhcp pool FINANCE
 network 10.10.20.0 255.255.255.0
 default-router 10.10.20.1
 dns-server 10.10.30.10

ip dhcp pool IT
 network 10.10.40.0 255.255.255.0
 default-router 10.10.40.1
 dns-server 10.10.30.10

ip dhcp pool GUEST
 network 10.10.50.0 255.255.255.0
 default-router 10.10.50.1
 dns-server 10.10.30.10

ip access-list extended USERS-IN
 remark Allow DHCP client requests to the router
 permit udp any eq bootpc any eq bootps
 remark Allow DNS and HTTP to the internal service host
 permit udp 10.10.10.0 0.0.0.255 host 10.10.30.10 eq domain
 permit tcp 10.10.10.0 0.0.0.255 host 10.10.30.10 eq domain
 permit tcp 10.10.10.0 0.0.0.255 host 10.10.30.10 eq 80
 remark Deny all other access to internal lab networks
 deny ip 10.10.10.0 0.0.0.255 10.10.0.0 0.0.255.255
 remark Permit non-internal destinations
 permit ip 10.10.10.0 0.0.0.255 any

ip access-list extended GUEST-IN
 remark Allow DHCP client requests to the router
 permit udp any eq bootpc any eq bootps
 remark Optional lab visibility: allow echo to the local gateway
 permit icmp 10.10.50.0 0.0.0.255 host 10.10.50.1 echo
 remark Allow DNS only to the internal DNS host
 permit udp 10.10.50.0 0.0.0.255 host 10.10.30.10 eq domain
 permit tcp 10.10.50.0 0.0.0.255 host 10.10.30.10 eq domain
 remark Deny other internal access
 deny ip 10.10.50.0 0.0.0.255 10.10.0.0 0.0.255.255
 remark Permit non-internal destinations
 permit ip 10.10.50.0 0.0.0.255 any

access-list 1 permit 10.10.0.0 0.0.255.255
ip nat inside source list 1 interface gigabitEthernet0/1 overload

ip domain-name enterprise.lab
username admin privilege 15 secret LabOnly!2026
enable secret LabEnable!2026
! In Packet Tracer, "crypto key generate rsa" may prompt for the modulus.
crypto key generate rsa modulus 1024
ip ssh version 2

access-list 40 permit 10.10.40.0 0.0.0.255
line vty 0 4
 transport input ssh
 login local
 access-class 40 in
 exec-timeout 5 0

end
copy running-config startup-config
